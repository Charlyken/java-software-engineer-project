@startuml

class Utilisateur {
  id : Long
  nom : String
  email : String
  role : Role
}

enum Role {
  ADMIN
  USER
}

class Etudiant {
  promotion : String
}

class Administrateur {
  role : String
}

class Dominante {
  id : Long
  nom : String
  description : String
}

class Session {
  id : Long
  date : Date
  heureDebut : Time
  heureFin : Time
  capaciteMax : int
}

class Campagne {
  id : Long
  dateOuverture : DateTime
  dateFermeture : DateTime
  etat : EtatCampagne
}

enum EtatCampagne {
  PREPARATION
  OUVERTE
  FERMEE
  TRAITEE
  ARCHIVEE
}


class Choix {
  listeChoix : ArrayList
  dateSaisie : DateTime
}

enum Statut {
  EN_ATTENTE
  CONFIRME
  REFUSE
}

class Inscription {
  id: Long
  dateInscription : DateTime
  dateValidation : DateTime
  choix : Choix
  statut : Statut
}

@enduml
